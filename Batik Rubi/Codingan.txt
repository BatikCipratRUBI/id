    <script>
        /* --- GLOBAL VARIABEL --- */
        let diskonAktif = 0; 
        let namaVoucherAktif = "";

        /* 1. FUNGSI MENU */
        function toggleMenu() {
            document.getElementById('sidebar').classList.toggle('active');
            document.getElementById('overlay').classList.toggle('active');
        }

        /* 2. FUNGSI QTY */
        function tambahQty(id) {
            let input = document.getElementById(id);
            input.value = parseInt(input.value) + 1;
        }
        function kurangQty(id) {
            let input = document.getElementById(id);
            if (parseInt(input.value) > 1) {
                input.value = parseInt(input.value) - 1;
            }
        }

        /* 3. FUNGSI CEK VOUCHER (PROMO KKN / DESA) */
        function cekVoucherGlobal() {
            let input = document.getElementById('globalCode');
            let pesan = document.getElementById('pesanStatus');
            let kode = input.value.toUpperCase().trim();

            diskonAktif = 0;
            namaVoucherAktif = "";
            pesan.style.color = "red";

            // LOGIKA DISKON BATIK
            if (kode === "PROMORUBI") {
                diskonAktif = 10000; 
                namaVoucherAktif = "PROMORUBI";
                pesan.style.color = "green";
                pesan.innerText = "Voucher Berhasil! Diskon Rp 10.000/item aktif.";
            }
            else if (kode === "KKNKEDUNGDOWO") {
                diskonAktif = 20000; 
                namaVoucherAktif = "KKNKEDUNGDOWO";
                pesan.style.color = "green";
                pesan.innerText = "Voucher Spesial KKN Berhasil! Diskon Rp 20.000/item.";
            }
            else {
                pesan.innerText = "Kode Voucher tidak valid.";
                updateTampilanSemuaProduk(); 
                return;
            }
            updateTampilanSemuaProduk();
        }

        /* 4. FUNGSI VISUAL: CORET HARGA */
        function updateTampilanSemuaProduk() {
            // Karena hanya ada 4 produk saat ini, loop dari 1 s/d 4
            for (let i = 1; i <= 4; i++) {
                let elOri = document.getElementById('ori-' + i);
                let elFinal = document.getElementById('final-' + i);

                if (elOri && elFinal) {
                    let textHarga = elOri.innerText.replace("Rp ", "").replace(/\./g, "");
                    let hargaAsli = parseInt(textHarga);

                    if (diskonAktif > 0) {
                        let hargaBaru = hargaAsli - diskonAktif;
                        elOri.classList.add('coret');
                        elFinal.innerText = "Rp " + hargaBaru.toLocaleString('id-ID');
                        elFinal.classList.add('show');
                    } else {
                        elOri.classList.remove('coret');
                        elFinal.classList.remove('show');
                        elFinal.innerText = "";
                    }
                }
            }
        }

        /* 5. FUNGSI CHECKOUT WA */
        function beliProduk(nama, hargaAsli, idQty) {
            let qty = parseInt(document.getElementById(idQty).value);
            let hargaSatuanBersih = hargaAsli - diskonAktif;
            if (hargaSatuanBersih < 0) hargaSatuanBersih = 0;
            let totalAkhir = hargaSatuanBersih * qty;

            const nomorWA = "628123456789"; // UBAH NOMOR WA INI KE NOMOR BUMDES / PENGRAJIN
            
            let pesan = `Halo Admin Batik RUBI, saya ingin memesan:\n\n`;
            pesan += `*${qty}x ${nama}*\n`;
            
            if (diskonAktif > 0) {
                pesan += `(Menggunakan Voucher: ${namaVoucherAktif})\n`;
                pesan += `Harga Normal: Rp ${(hargaAsli * qty).toLocaleString('id-ID')}\n`;
                pesan += `Potongan: - Rp ${(diskonAktif * qty).toLocaleString('id-ID')}\n`;
            }

            pesan += `\n*TOTAL BAYAR: Rp ${totalAkhir.toLocaleString('id-ID')}*\n\n`;
            pesan += `Mohon info untuk ketersediaan stok dan biaya ongkir ke alamat saya ya. Terima kasih!`;

            window.open(`https://wa.me/${nomorWA}?text=${encodeURIComponent(pesan)}`, '_blank');
        }
    </script>
